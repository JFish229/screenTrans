cmake_minimum_required(VERSION 3.24)
project(FlexMeetASR LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(ASR_ENGINE_ONLY "Build dependency-free segmentation tests only" OFF)
option(ASR_PREBUILT_WHISPER "Use the checked-in CPU libraries (Windows x64 only)" OFF)
set(ASR_WHISPER_SOURCE "" CACHE PATH "Local whisper.cpp source at the locked revision")
include(CTest)
if(BUILD_TESTING)
    add_executable(asr_engine_test tests/stream_engine_test.cpp)
    add_test(NAME stream_engine COMMAND asr_engine_test)
    add_executable(asr_audio_test tests/audio_mix_test.cpp)
    add_test(NAME audio_routing COMMAND asr_audio_test)
    add_library(asr_c_header OBJECT tests/c_header.c)
    target_include_directories(asr_c_header PRIVATE include)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/metrics_test.py")
    endif()
endif()
if(ASR_ENGINE_ONLY)
    return()
endif()
if(ASR_PREBUILT_WHISPER)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Bundled libraries require Windows x64")
    endif()
    add_library(asr_whisper INTERFACE)
    target_include_directories(asr_whisper INTERFACE third/whisper.cpp/whisper_include third/whisper.cpp/ggml_include)
    foreach(lib whisper ggml-base ggml-cpu ggml)
        if(lib STREQUAL "whisper")
            set(folder whisper)
        else()
            set(folder ggml)
        endif()
        target_link_libraries(asr_whisper INTERFACE
            debug "${CMAKE_CURRENT_SOURCE_DIR}/third/whisper.cpp/${folder}_debug/${lib}.lib"
            optimized "${CMAKE_CURRENT_SOURCE_DIR}/third/whisper.cpp/${folder}_release/${lib}.lib")
    endforeach()
    find_package(OpenMP REQUIRED)
    target_link_libraries(asr_whisper INTERFACE OpenMP::OpenMP_CXX)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
    if(ASR_WHISPER_SOURCE)
        add_subdirectory("${ASR_WHISPER_SOURCE}" whisper)
    else()
        include(FetchContent)
        FetchContent_Declare(whisper
            URL https://codeload.github.com/ggml-org/whisper.cpp/zip/6e4ab854f67f743900934a703d5603419384c961
            URL_HASH SHA256=777504793d609419aea6121b5865702b22755c54b82f9c22c6e4973ce19019ae)
        FetchContent_MakeAvailable(whisper)
    endif()
    add_library(asr_whisper INTERFACE)
    target_link_libraries(asr_whisper INTERFACE whisper)
endif()
add_library(plugin_ASR SHARED src/asr.cpp src/stream_api.cpp)
target_include_directories(plugin_ASR PUBLIC include)
target_link_libraries(plugin_ASR PRIVATE asr_whisper)
if(BUILD_TESTING)
    add_executable(asr_worker_test tests/stream_worker_test.cpp src/asr.cpp)
    target_include_directories(asr_worker_test PRIVATE include)
    target_link_libraries(asr_worker_test PRIVATE asr_whisper)
    add_test(NAME stream_worker COMMAND asr_worker_test)
endif()
set_target_properties(plugin_ASR PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/plugin")
if(MSVC)
    target_compile_options(plugin_ASR PRIVATE /utf-8 /W4)
endif()
